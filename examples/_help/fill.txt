command: affinity-axi fill
summary: Fill a saved .af by element name from JSON or CSV: one file per record
args[1]{name,required,description}:
  template,yes,the saved .af whose fillable elements are named slot:<key> in the Layers panel
flags[15]{flag,default,description}:
  --data,,"records: a .json array of objects, or a .csv with a header row (required). Each field fills the element named slot:<field>; an empty value blanks a text slot and keeps its style, a field the record does not have leaves the slot as the template has it. A picture slot may be a path to .png .jpg .jpeg .webp .heic .heif .tif or .tiff (JPEG and TIFF with EXIF orientation are placed upright) or {""image"": ""photo.jpg"", ""cutout"": true|""subject"", ""colourise"": true, ""align"": ""centre""|""fit""|""none"" (default centre with cutout), ""padding"": px (with fit)} (JSON only)"
  --out-dir,,folder under Desktop: each record is saved as <dir>/<key>.af (required)
  --key,,"record field that names each record (default: the record index); made filesystem-safe, must be unique"
  --export,,also export each record with this preset (see: affinity-axi presets) as <dir>/<key>.<ext>; Affinity's XMP (account name) is stripped from PNG/JPEG/WebP and the kit's `credit` is embedded when set (needs exiftool)
  --all-artboards,,"with --export: one file per artboard, <dir>/<key>-<artboard>.<ext>"
  --max-kb,,"largest file size in KB; steps JPEG/WebP/PNG presets down, then size (to --min-scale)"
  --min-scale,,smallest size step for --max-kb as a fraction of the requested size (default 0.5)
  --limit,,process only the first N records
  --continue-on-error,,keep going after a failed record (default: stop at the first failure); a failed ML record whose undo leaves the document dirty is saved under _failed/ and closed
  --force,,overwrite a .af that was changed outside affinity-axi (stale manifest) or that affinity-axi never saved (no manifest)
  --ignore-unknown,,a data key with no slot is skipped instead of failing the record
  --brand,,"brand kit: a .brand.json path or a name under ./brands/ (default $AFFINITY_AXI_BRAND, then ./brand.json)"
  --timeout,60000,per-script timeout in milliseconds (ML options cutout and colourise take about 10 s per photo)
  --url,,Affinity MCP URL (AFFINITY_MCP_URL wins; default http://[::1]:6767)
  --help,,show this help
examples[2]:
  affinity-axi fill post.af --data posts.json --key slug --out-dir ~/Desktop/out --export PNG
  affinity-axi fill post.af --data posts.csv --key slug --out-dir ~/Desktop/out --limit 2 --ignore-unknown
output[2]:
  records[2]{key,status,doc,ms,note}:
  exported: 2
